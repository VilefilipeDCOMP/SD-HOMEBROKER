const WS_LOCAL = location.protocol === "file:" || ["5173", "5500"].includes(location.port);
const WS_PROTOCOL = location.protocol === "https:" ? "wss:" : "ws:";
const WS_HOST = WS_LOCAL ? `${location.hostname || "localhost"}:8086` : location.host;
const WS_URL = `${WS_PROTOCOL}//${WS_HOST}/ws`;
const AUTH_BASE = WS_LOCAL ? `${WS_PROTOCOL === "wss:" ? "https:" : "http:"}//${WS_HOST}` : "";

// desenar as telas
function desenharSaldo() {
  $("saldo").textContent = "R$ " + fmt(estado.saldo);
}

function desenharTabela() {
  $("tabela").innerHTML = estado.ativos
    .map((a) => {
      const v = ((a.preco - a.base) / a.base) * 100;
      const tipo = v > 0.005 ? "alta" : v < -0.005 ? "baixa" : "neutro";
      const sinal = tipo === "alta" ? "+" : tipo === "baixa" ? "âˆ’" : "";
      return `
        <div class="linha">
          <span class="codigo">${a.codigo}<small class="nome-empresa">${a.nome ? ` Â· ${a.nome}` : ""}</small></span>
          <span class="preco">R$ ${fmt(a.preco)}</span>
          <span class="variacao ${tipo}">${sinal}${fmt(Math.abs(v))}%</span>
        </div>`;
    })
    .join("");
}

function desenharBoleta() {
  const codigo = $("ativo").value;
  const pos = estado.carteira[codigo] ?? 0;
  const ativo = achar(codigo);
  $("posicao").textContent = `VocÃª possui: ${pos} ${codigo}`;
  $("preco-mercado").textContent = ativo
    ? `PreÃ§o de mercado agora: R$ ${fmt(ativo.preco)}`
    : "";

  $("feed").innerHTML = estado.logs
    .map((l) => `<div class="feed-item"><time>${l.hora}</time><span class="${l.tipo}">${l.msg}</span></div>`)

// ponto unico de entrada de preco
function aplicarPreco(codigo, novoPreco, nome) {
  let ativo = achar(codigo);
  if (!ativo) {
    ativo = { codigo, nome: nome || "", preco: novoPreco, base: novoPreco };
    estado.ativos.push(ativo);
    if (estado.carteira[codigo] === undefined) estado.carteira[codigo] = 0;
    atualizarSelectAtivos();
  } else if (nome && !ativo.nome) {
    ativo.nome = nome;
  }
  if (novoPreco === ativo.preco) return;
  ativo.preco = novoPreco;
  desenharTabela();
  desenharBoleta();
  log(`PreÃ§o da ${codigo} atualizou para R$ ${fmt(novoPreco)}.`);
}

// --- autenticacao REST ---

// conta fixa para demonstracao quando o backend nao esta disponivel
const CONTA_TESTE = { username: "teste", password: "teste123" };

// erro que indica backend fora do ar (ou rota /auth ainda nao publicada)
function erroOffline() {
  const e = new Error("Servidor indisponÃ­vel");
  e.offline = true;
  return e;
}

// faz login ou cadastro no backend e retorna { token, username, ... }
async function authRest(endpoint, username, password) {
  let res;
  try {
    res = await fetch(`${AUTH_BASE}${endpoint}`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ username, password }),
    });
  } catch {
    throw erroOffline(); // falha de rede / CORS / servidor desligado
  }
  if (res.status === 404) throw erroOffline(); // backend sem o modulo de auth
  // corpo pode vir vazio (ex.: 404 sem payload), entao nao quebra se nao for JSON
  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    // contrato do M2: { type: "AUTH_ERROR", code, message }
    throw new Error(data.message || data.error || `Erro ${res.status}`);
  }
  return data;
}

function authLocal(endpoint, username, password) {
  const contas = contasLocais();
  if (endpoint === "/auth/register") {
    if (!username || !password) throw new Error("Username e senha sÃ£o obrigatÃ³rios");
    if (contas[username] || username === CONTA_TESTE.username) {
      throw new Error("Username jÃ¡ estÃ¡ em uso");
    }
    contas[username] = password;
    localStorage.setItem("contasDemo", JSON.stringify(contas));
  } else {
    const ok =
      (username === CONTA_TESTE.username && password === CONTA_TESTE.password) ||
      (contas[username] !== undefined && contas[username] === password);
    if (!ok) throw new Error("UsuÃ¡rio ou senha invÃ¡lidos");
  }
  return {
    token: null,
    username,
    balance: 10000,
    portfolio: { PETR4: 0, VALE3: 0, ITUB4: 0, MGLU3: 0 },
  };
}

// tenta o backend; se ele estiver fora, usa o modo demonstracao
async function autenticar(endpoint, username, password) {
  try {
    return await authRest(endpoint, username, password);
  } catch (e) {
    if (!e.offline) throw e; // erro real do backend (senha errada, usuario existente...)
    return authLocal(endpoint, username, password);
  }
}

let stomp = null;
let timerPrecos = null;

// recebe o token JWT e abre a conexÃ£o STOMP autenticada
function conectarStomp(token) {
  stomp = new StompJs.Client({
    brokerURL: WS_URL,
    connectHeaders: {
      // token JWT no header do CONNECT â€” backend valida aqui, nao na URL
      Authorization: `Bearer ${token}`,
    },
    reconnectDelay: 5000,

    onConnect: () => {
      log("Conectado ao servidor do mercado.");

      // para a simulacao local se o server conectou
      if (timerPrecos) {
        clearInterval(timerPrecos);
        timerPrecos = null;
      }

      // topico publico de mercado
      stomp.subscribe("/topic/market", (msg) => {
        try {
          const evento = JSON.parse(msg.body);
          const d = evento.data || evento;
          const cod = d.symbol || d.ticker || d.codigo || d.assetCode;
          const preco = d.currentPrice ?? d.preco_atual ?? d.price ?? d.preco;
          const nome = d.name || d.nome || d.nome_empresa || "";
          if (cod && preco != null) {
            aplicarPreco(cod, Number(preco), nome);
          }
        } catch (e) {
          console.error("Erro ao ler evento do mercado", e);
        }
      });

      // respostas individuais (dados de saldo, carteira ou lista de ativos)
      stomp.subscribe("/user/queue/replies", (msg) => {
        try {
          const resp = JSON.parse(msg.body);
          const d = resp.data || resp;
          const tipo = resp.type || "";

          // 1. CatÃ¡logo oficial de ativos (ASSET_LIST)
          const listaAtivos = Array.isArray(d) ? d : (Array.isArray(resp.ativos) ? resp.ativos : null);
          if (listaAtivos && (tipo === "ASSET_LIST" || !tipo)) {
            listaAtivos.forEach((a) => {
              const cod = a.symbol || a.ticker || a.codigo;
              const p = a.price ?? a.currentPrice ?? a.preco_atual ?? a.preco;
              const nome = a.name || a.nome || a.nome_empresa || "";
              if (cod && p != null) aplicarPreco(cod, Number(p), nome);
            });
            log("CatÃ¡logo oficial de ativos sincronizado.", "info");
          }

          // 2. HistÃ³rico de ordens (ORDER_HISTORY)
          if (tipo === "ORDER_HISTORY" && Array.isArray(d)) {
            log(`HistÃ³rico sincronizado: ${d.length} ordem(ns) registrada(s).`, "info");
          }

          if (tipo === "ORDER_STATUS" && d) {
            log(`Status da ordem ${d.orderId ? d.orderId.substring(0, 8) : ""}: ${d.status}`, d.status === "ACCEPTED" ? "ok" : "info");
          }

          // 4. AtualizaÃ§Ãµes de saldo ou carteira
          const saldo = d.saldo ?? d.balance ?? resp.saldo ?? resp.balance;

          if (carteira) {

          const ehSucesso = status === "ACCEPTED" || status === "OK";

            desenharSaldo();
          }
          const carteira = d.carteira ?? d.portfolio;
          if (carteira) {
            Object.assign(estado.carteira, carteira);
            desenharBoleta();
          }
        } catch (e) {

//Ã© o controller das ordens
//o @Controller Ã© o aviso ao spring que vai ser roteamento de msg STOMP
@RequiredArgsConstructor
@Controller
public class OrderController {

    private static final Logger log = LoggerFactory.getLogger(OrderController.class);

    private final OrderService orderService;
    private final CurrentUserResolver users;

    //Ã© por onde a ordem chega
    @MessageMapping("/orders/create")
    //envia a resposta somente para o user que enviou a ordem
    @SendToUser(destinations = "/queue/orders", broadcast = false)
    public MarketEvent<OrderResult> submit(SubmitOrderMessage message, SimpMessageHeaderAccessor headers) {
        String userId = users.resolve(headers);//pega o user
        SubmitOrderMessage body = message != null ? message //validaÃ§Ã£o caso a ordem seja toda vazia
                : new SubmitOrderMessage(null, null, null, null, null);

        //processa a ordem lÃ¡ no orderSertvice
        OrderResult result = orderService.submit(body.toRequest(userId));
        //e retorna a confirmaÃ§Ã£o
        return event("ORDER_ACK", result.clientOrderId(), result);
    }

    //consulta o status atual de uma ordem
    @MessageMapping("/orders/status")
    @SendToUser(destinations = "/queue/replies", broadcast = false)
    public MarketEvent<OrderResult> status(OrderStatusMessage message, SimpMessageHeaderAccessor headers) {
        String userId = users.resolve(headers);
        String orderId = message != null ? message.orderId() : null;

        return orderService.findByIdForUser(orderId, userId)
                .map(order -> event("ORDER_STATUS", order.clientOrderId(), OrderResult.of(order)))
                .orElseGet(() -> event("ORDER_NOT_FOUND", orderId, null));
    }

    //historico de ordens do user, ordenada das mais antigas para as mais recentes
    @MessageMapping("/orders/mine")
    @SendToUser(destinations = "/queue/replies", broadcast = false)
    public MarketEvent<List<OrderResult>> myOrders(SimpMessageHeaderAccessor headers) {
        String userId = users.resolve(headers);
        List<OrderResult> orders = orderService.findByUser(userId).stream()
                .map(OrderResult::of)
                .toList();
        return event("ORDER_HISTORY", null, orders);
    }

    //exception para caso o usuario tenha enviado json quebrado ou algo assim
    @MessageExceptionHandler
    @SendToUser(destinations = "/queue/orders", broadcast = false)
    public MarketEvent<OrderResult> handleMalformedMessage(Exception exception) {
        log.warn("ORDER_MALFORMED_MESSAGE {}", exception.toString());
        //retorna a msg de dados invÃ¡lidos
        return event("ORDER_ACK", null, OrderResult.rejected(null, RejectReason.INVALID_FIELDS));
    }

    //marca o evento com o data hora e id do evento
    private static <T> MarketEvent<T> event(String type, String requestId, T data) {
        return new MarketEvent<>(type, requestId, UUID.randomUUID(), Instant.now(), data);
    }
}

package br.ufs.simulador_bolsa.orders;

import java.math.BigDecimal;
import java.util.Locale;

//Ã© o corpo JSON que envia
public record SubmitOrderMessage(
        String clientOrderId,
        String symbol,
        String side,
        Long quantity,
        String price
) {

    //deixa salvo no record temporario
    public OrderRequest toRequest(String userId) {
        return new OrderRequest(userId, clientOrderId, symbol, parseSide(side), quantity, parsePrice(price));
    }

    //o restante sÃ£o as validaÃ§Ãµes
    private static OrderSide parseSide(String value) {
        if (value == null) {
            return null;
        }
        try {
            return OrderSide.valueOf(value.trim().toUpperCase(Locale.ROOT));
        } catch (IllegalArgumentException e) {
            return null;
        }
    }

    private static BigDecimal parsePrice(String value) {
        if (value == null) {
            return null;
        }
        try {
            return new BigDecimal(value.trim().replace(',', '.'));
        } catch (NumberFormatException e) {
            return null;
        }
    }
}

CREATE TABLE IF NOT EXISTS historico_negociacoes (
    id BIGSERIAL PRIMARY KEY,
    ordem_compra_id BIGINT NOT NULL REFERENCES ordens(id),
    ordem_venda_id BIGINT NOT NULL REFERENCES ordens(id),
    ativo_id BIGINT NOT NULL REFERENCES ativos(id),
    quantidade_executada INT NOT NULL CHECK (quantidade_executada > 0),
    preco_executado NUMERIC(15, 2) NOT NULL CHECK (preco_executado > 0),
    executado_em TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);
